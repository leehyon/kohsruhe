{{- $src := .Get "src" -}} {{- $thumb := $src | default (printf "%s." (.Get "thumb") | replace (.Get "link") ".") -}} {{/* Resolve image dimensions: explicit > auto-detected > unknown */}} {{- $userW := int (.Get "width" | default 0) -}} {{- $userH := int (.Get "height" | default 0) -}} {{- $hasUserDims := and (gt $userW 0) (gt $userH 0) -}} {{- $detectedW := 0 -}} {{- $detectedH := 0 -}} {{- if and $src (not $hasUserDims) -}} {{/* Build-time fetch to read intrinsic image dimensions. resources.GetRemote populates Width/Height from response headers / first bytes; the result is internally cached, so we don't need resources.CopyRemote here. */}} {{- with try (resources.GetRemote $src) -}} {{- with .Err -}} {{/* swallow: leave $detectedW/H at 0, runtime fallback kicks in */}} {{- else with .Value -}} {{- if .Width -}} {{- $detectedW = .Width -}} {{- $detectedH = .Height -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $finalW := cond $hasUserDims $userW $detectedW -}} {{- $finalH := cond $hasUserDims $userH $detectedH -}} {{- $hasDims := and (gt $finalW 0) (gt $finalH 0) -}} {{- if not ($.Page.Scratch.Get "figurecount") }}{{ end }} {{- $.Page.Scratch.Add "figurecount" 1 }}
